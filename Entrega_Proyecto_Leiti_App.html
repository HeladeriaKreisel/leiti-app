// Clase de Dominio: Finca — Calcula ocupación del tanque, meta diaria y promedio por vaca
export class Finca {
  constructor(data = {}) {
    this.id = data.id || null;
    this.nombre_finca = data.nombre_finca || 'Hacienda La Pradera';
    this.propietario_nombre = data.propietario_nombre || 'TOMAS GARCIA';
    this.capacidad_tanque_litros = Number(data.capacidad_tanque_litros ?? 500);
    this.vacas_en_ordeno = Number(data.vacas_en_ordeno ?? 28);
    this.meta_diaria_litros = Number(data.meta_diaria_litros ?? 150);
  }
  calcularPorcentajeOcupacionTanque(litrosActuales) {
    if (!this.capacidad_tanque_litros || this.capacidad_tanque_litros <= 0) return 0;
    return Math.min(100, Math.round((Number(litrosActuales) / this.capacidad_tanque_litros) * 100));
  }
}

// Clase de Dominio: RegistroLeche — Liquidación financiera de cada ordeño
export class RegistroLeche {
  constructor(data = {}) {
    this.fecha = data.fecha || new Date().toISOString().split('T')[0];
    this.turno = data.turno || 'morning';
    this.litros = Number(data.litros ?? 0);
    this.precio_litro = Number(data.precio_litro ?? 2200);
    this.flete_por_litro = Number(data.flete_por_litro ?? 50);
  }
  calcularSubtotal() { return this.litros * this.precio_litro; }
  calcularDescuentoFlete() { return this.litros * this.flete_por_litro; }
  calcularNeto() { return this.calcularSubtotal() - this.calcularDescuentoFlete(); }
}

export class Cliente {
  constructor(data = {}) {
    this.id = data.id || null;
    this.nombre = data.nombre || '';
    this.empresa = data.empresa || data.nombre || '';
    this.contacto_nombre = data.contacto_nombre || 'Encargado de Compras';
    this.telefono = data.telefono || '';
    this.precio_acordado_litro = Number(data.precio_acordado_litro ?? 2200);
    this.ruta = data.ruta || 'Ruta General';
  }
  obtenerEnlaceWhatsApp(mensaje = '') {
    const texto = encodeURIComponent(mensaje || `Hola ${this.contacto_nombre}, despacho Leiti.app`);
    return `https://wa.me/${this.telefono_wa}?text=${texto}`;
  }
}

export class VentaRemision {
  constructor(data = {}) {
    this.numero_remision = data.numero_remision || `REM-${String(Math.floor(100 + Math.random() * 900)).padStart(4, '0')}`;
    this.cliente_nombre = data.cliente_nombre || '';
    this.litros = Number(data.litros ?? 0);
    this.precio_litro = Number(data.precio_litro ?? 2200);
    this.estado = data.estado || 'paid';
  }
  calcularTotal() { return this.litros * this.precio_litro; }
  alternarEstadoPago() { return (this.estado = this.estado === 'paid' ? 'pending' : 'paid'); }
}
src/App.jsx — Componente Raíz, Estado Reactivo y Sincronización React 18 Hooks 
export default function App() {
  const [currentTab, setCurrentTab] = useState('inicio');
  const [finca, setFinca] = useState(null);
  const [registros, setRegistros] = useState([]);
  const [clientes, setClientes] = useState([]);
  const [ventas, setVentas] = useState([]);
  const [isSyncing, setIsSyncing] = useState(false);

  const loadData = async () => {
    setIsSyncing(true);
    const [fincaRes, registrosRes, clientesRes, ventasRes] = await Promise.all([
      getFincaData(), getRegistrosLeche(), getClientes(), getVentas()
    ]);
    if (fincaRes?.data) setFinca(fincaRes.data);
    if (registrosRes?.data) setRegistros(registrosRes.data);
    if (clientesRes?.data) setClientes(clientesRes.data);
    if (ventasRes?.data) setVentas(ventasRes.data);
    setIsSyncing(false);
  };

  useEffect(() => { loadData(); }, []);
  // Renderizado condicional de DashboardView, MiFincaView, LecheDiaView, VentasView y ClientesView
}

const SUPABASE_URL = "https://cthvxtjwstyezwzuohdx.supabase.co";
async function sbRest(path, options = {}) {
  const response = await fetch(`${SUPABASE_URL}/rest/v1/${path}`, {
    method: options.method || 'GET',
    headers: {
      'apikey': SUPABASE_ANON_KEY,
      'Authorization': `Bearer ${SUPABASE_ANON_KEY}`,
      'Content-Type': 'application/json',
      'Prefer': options.prefer || 'return=representation'
    },
    body: options.body ? JSON.stringify(options.body) : undefined
  });
  return await response.json();
}
